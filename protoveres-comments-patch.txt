function renderPostComments(postId,comments){
  const list=document.querySelector(`[data-comments-list="${CSS.escape(postId)}"]`);
  const btn=document.querySelector(`[data-social-comments-label="${CSS.escape(postId)}"]`);
  if(btn)btn.textContent=`💬 Bình luận${comments.length?` (${comments.length})`:''}`;
  if(!list)return;

  if(!comments.length){
    list.innerHTML='<div class="postCommentEmpty">Chưa có bình luận. Hãy là người đầu tiên 💬</div>';
    return;
  }

  list.innerHTML=comments.map(c=>{
    const name=c.authorName||'Protoveres User';
    const av=c.authorAvatarURL||'';
    const del=user?.uid===c.uid
      ? `<button class="postCommentDelete" type="button" data-comment-delete="${esc(c.id)}">Xoá</button>`
      : '';

    return `<div class="postComment">
      <div class="postCommentAvatar">${av?`<img src="${esc(av)}" alt="">`:esc(initials(name))}</div>
      <div class="postCommentBody">
        <b>${esc(name)}</b>
        <div class="postCommentText">${esc(c.text||'')}</div>
        <div class="postCommentMeta">
          <span>${esc(socialDate(c.createdAt))}</span>${del}
        </div>
      </div>
    </div>`;
  }).join('');

  list.querySelectorAll('[data-comment-delete]').forEach(b=>{
    b.onclick=()=>deletePostComment(postId,b.dataset.commentDelete);
  });
}

function initPostComments(postId){
  if(socialCommentUnsubs[postId])return;

  const list=document.querySelector(`[data-comments-list="${CSS.escape(postId)}"]`);
  if(list)list.innerHTML='<div class="postCommentEmpty">Đang tải bình luận…</div>';

  const q=query(
    collection(db,'posts',postId,'comments'),
    orderBy('createdAt','asc')
  );

  socialCommentUnsubs[postId]=onSnapshot(q,snap=>{
    const comments=[];
    snap.forEach(d=>comments.push({id:d.id,...d.data()}));
    renderPostComments(postId,comments);
  },e=>{
    if(list){
      list.innerHTML=
        `<div class="postCommentEmpty">Không tải được bình luận: ${esc(friendlyError(e))}</div>`;
    }
  });
}

function closePostComments(postId){
  if(socialCommentUnsubs[postId]){
    try{socialCommentUnsubs[postId]()}catch{}
    delete socialCommentUnsubs[postId];
  }
}

function togglePostComments(postId){
  const panel=document.querySelector(`[data-comment-panel="${CSS.escape(postId)}"]`);
  const btn=document.querySelector(`[data-social-comments="${CSS.escape(postId)}"]`);
  if(!panel)return;

  const open=panel.classList.toggle('open');
  if(btn)btn.classList.toggle('active',open);

  if(open)initPostComments(postId);
  else closePostComments(postId);
}

async function submitPostComment(postId){
  if(!user)return;

  const input=document.querySelector(`[data-comment-input="${CSS.escape(postId)}"]`);
  const send=document.querySelector(`[data-comment-send="${CSS.escape(postId)}"]`);
  if(!input)return;

  const text=input.value.trim();
  if(!text)return;

  if(text.length>1000){
    alert('Bình luận tối đa 1000 ký tự.');
    return;
  }

  if(send){
    send.disabled=true;
    send.textContent='…';
  }

  try{
    const prof=await getMyProfile();

    const commentRef=await addDoc(
      collection(db,'posts',postId,'comments'),
      {
        uid:user.uid,
        authorName:
          prof?.username||
          prof?.displayName||
          user.displayName||
          'Protoveres User',
        authorAvatarURL:
          prof?.avatarURL||
          prof?.photoURL||
          user.photoURL||
          null,
        text,
        createdAt:serverTimestamp()
      }
    );

    input.value='';

    const post=socialPostsCache.find(x=>x.id===postId);

    if(post?.ownerUid&&post.ownerUid!==user.uid){
      createUserNotification(post.ownerUid,{
        type:'post_comment',
        postId,
        targetType:'post',
        message:text.slice(0,120),
        commentId:commentRef.id
      }).catch(()=>{});
    }

  }catch(e){
    alert(
      e?.code==='permission-denied'
        ? 'Firebase từ chối bình luận. Hãy kiểm tra Rules cho /posts/{postId}/comments.'
        : friendlyError(e)
    );
  }finally{
    if(send){
      send.disabled=false;
      send.textContent='Gửi';
    }
  }
}

async function deletePostComment(postId,commentId){
  if(!user||!commentId)return;

  try{
    await deleteDoc(
      doc(db,'posts',postId,'comments',commentId)
    );
  }catch(e){
    alert(friendlyError(e));
  }
}

KẾT QUẢ:
Bài đăng sẽ có:
👍 Like   ❤️ Thả tim   💬 Bình luận   ↗️ Chia sẻ

Bấm "💬 Bình luận":
- Mở khung bình luận realtime.
- Gửi comment vào posts/{postId}/comments/{commentId}.
- Enter để gửi, Shift+Enter để xuống dòng.
- Chủ comment xoá được comment của mình.
- Chủ bài đăng nhận notification + push "X đã bình luận bài đăng của bạn".
